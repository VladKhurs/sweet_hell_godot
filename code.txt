		player.set_controls_enabled(true)

		$RhythmEngine.play_music()

>>!>FILEPATH: ./levels/templates/novel_level/novel_level.gd
extends Control
class_name NovelLevel

@export_category("Story & Progression")
@export var novel_id: String = ""
@export var dialogue_resource: DialogueResource
@export var dialogue_title: String = "start"
@export_file("*.tscn") var next_level_path: String = ""
@export var skip_if_already_read: bool = false

@export_category("Visuals")
@export_dir var images_folder: String = "res://assets/images/"
@export var default_background: Texture2D
@export var transition_duration: float = 0.4

@onready var bg_current: TextureRect = $BackgroundCurrent
@onready var bg_next: TextureRect = $BackgroundNext

var _fade_tween: Tween
var _current_bg_key: String = ""

func _ready() -> void:
	set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)

	if skip_if_already_read and not novel_id.is_empty() and GameState.get_world_state(novel_id, "completed") == true:
		_go_to_next_level()
		return

	if default_background:
		bg_current.texture = default_background

	if not dialogue_resource:
		push_error("[Novel] dialogue_resource не назначен!")
		return

	DialogueManager.got_dialogue.connect(_on_got_dialogue)
	DialogueManager.dialogue_ended.connect(_on_dialogue_ended)
	DialogueManager.show_dialogue_balloon(dialogue_resource, dialogue_title, [self])

func set_bg(image_name: String) -> void:
	if image_name.is_empty() or image_name == _current_bg_key: return
	var tex := _find_texture(image_name)
	if not tex:
		push_error("[Novel] Картинка '%s' не найдена!" % image_name)
		return

	_current_bg_key = image_name
	if transition_duration <= 0.0:
		bg_current.texture = tex
		return

	if _fade_tween and _fade_tween.is_valid():
		_fade_tween.kill()

	bg_next.texture = tex
	bg_next.modulate.a = 0.0
	_fade_tween = create_tween()
	_fade_tween.tween_property(bg_next, "modulate:a", 1.0, transition_duration)
	_fade_tween.tween_callback(func():
		bg_current.texture = tex
		bg_next.modulate.a = 0.0
		bg_next.texture = null
	)

func _find_texture(file_name: String) -> Texture2D:
	var dirs: Array[String] = [images_folder, "res://assets/images/", "res://assets/"]
	if dialogue_resource:
		dirs.append(dialogue_resource.resource_path.get_base_dir())

	for d in dirs:
		var base_path: String = d.path_join(file_name)
		for ext: String in ["", ".png", ".webp", ".jpg"]:
			var full_path: String = base_path + ext
			if ResourceLoader.exists(full_path):
				return load(full_path) as Texture2D
	return null

func _on_got_dialogue(line: DialogueLine) -> void:
	if line.has_tag("bg"):
		set_bg(line.get_tag_value("bg"))

func _on_dialogue_ended(resource: DialogueResource) -> void:
	if resource != dialogue_resource: return
	if not novel_id.is_empty():
		GameState.set_world_state(novel_id, "completed", true)
	_go_to_next_level()

func _go_to_next_level() -> void:
	if next_level_path.is_empty(): return
	if DialogueManager.got_dialogue.is_connected(_on_got_dialogue):
		DialogueManager.got_dialogue.disconnect(_on_got_dialogue)
	if DialogueManager.dialogue_ended.is_connected(_on_dialogue_ended):
		DialogueManager.dialogue_ended.disconnect(_on_dialogue_ended)
	Game.load_level(next_level_path)

extends Node
class_name WalkCutscene

@export var walk_distance: float = 120.0
@export var walk_time: float = 2.5
@export var lumi: Node2D
@export var wall: Node2D

func _ready() -> void:
	await get_tree().process_frame
	var choice := GameState.story_point
	if choice.is_empty():
		return

	GameState.set_story_point("")
	_run_cutscene(choice)

func _run_cutscene(choice: String) -> void:
	var player = get_tree().get_first_node_in_group("player")
	if not player:
		player = get_tree().root.find_child("Player", true, false)
	if not player:
		return

	if not lumi: lumi = get_tree().root.find_child("Lumi", true, false)
	if not wall: wall = get_tree().root.find_child("Wall", true, false)

	player.set_controls_enabled(false)
	player.set_process(false)

	if wall:
		_set_wall_active(false)

	if choice == "walk-together":
		player.anim.flip_h = true
		player.anim.play("walk")
		if lumi and lumi.has_method("play_walk"): lumi.play_walk(true)

		var tween := create_tween().set_parallel(true)
		tween.tween_property(player, "global_position:x", player.global_position.x - walk_distance, walk_time)
		if lumi:
			tween.tween_property(lumi, "global_position:x", lumi.global_position.x - walk_distance, walk_time)
		await tween.finished

		player.anim.play("idle")
		if lumi and lumi.has_method("play_idle"): lumi.play_idle(true)
	else:
		if lumi and lumi.has_method("play_walk"): lumi.play_walk(true)
		if lumi:
			var tween := create_tween()
			tween.tween_property(lumi, "global_position:x", lumi.global_position.x - walk_distance - 50.0, walk_time)
			await tween.finished
			if lumi.has_method("play_idle"): lumi.play_idle(true)
		if wall:
			_set_wall_active(true)

	player.set_process(true)
	player.set_controls_enabled(true)

func _set_wall_active(active: bool) -> void:
	wall.visible = active
	var col = wall.get_node_or_null("CollisionShape2D")
	if col:
		col.set_deferred("disabled", not active)

>>!>FILEPATH: ./widgets/hud/touch/move_joystick.gd
extends Control
class_name MoveJoystick

@export var max_radius: float = 40.0
@export var deadzone: float = 0.15
@export var jump_threshold: float = 0.45
@export var jump_action: StringName = &"ui_accept"
@export var default_offset: Vector2 = Vector2(110.0, 110.0)

@export var base_color: Color = Color(1, 1, 1, 0.2)
@export var border_color: Color = Color(1, 1, 1, 0.5)
@export var knob_color: Color = Color(1, 1, 1, 0.6)

var touch_id: int = -1
var base_pos: Vector2
var knob_offset: Vector2 = Vector2.ZERO
var is_touched: bool = false

func _ready() -> void:
	mouse_filter = Control.MOUSE_FILTER_STOP
	base_pos = _get_default_pos()

func _notification(what: int) -> void:
	if what == NOTIFICATION_RESIZED and not is_touched:
		base_pos = _get_default_pos()
		queue_redraw()

func _get_default_pos() -> Vector2:
	return Vector2(default_offset.x, size.y - default_offset.y)

func _gui_input(event: InputEvent) -> void:
	if event is InputEventScreenTouch:
		if event.pressed and touch_id == -1:
			touch_id = event.index
			is_touched = true
			base_pos = event.position
			knob_offset = Vector2.ZERO
		elif not event.pressed and event.index == touch_id:
			_reset_joystick()
		queue_redraw()
	elif event is InputEventScreenDrag and event.index == touch_id:
		knob_offset = (event.position - base_pos).limit_length(max_radius)
		_apply_input(knob_offset / max_radius)
		queue_redraw()

func _draw() -> void:
	var center := base_pos if is_touched else _get_default_pos()
	draw_circle(center, max_radius, base_color)
	draw_arc(center, max_radius, 0, TAU, 32, border_color, 2.0)
	draw_circle(center + knob_offset, max_radius * 0.4, knob_color)

func _apply_input(vec: Vector2) -> void:
	var strength := clampf((vec.length() - deadzone) / (0.85 - deadzone), 0.0, 0.85) if vec.length() > deadzone else 0.0
	var norm := vec.normalized() * strength
	_set_axis("ui_left", "ui_right", norm.x)
	_set_axis("ui_up", "ui_down", norm.y)

	var should_jump := GameState.current_mode == &"platformer" and vec.y < -jump_threshold
	Input.action_press(jump_action) if should_jump else Input.action_release(jump_action)

func _set_axis(neg: StringName, pos: StringName, val: float) -> void:
	if val > 0.05:
		Input.action_press(pos, val)
		Input.action_release(neg)
	elif val < -0.05:
		Input.action_press(neg, -val)
		Input.action_release(pos)
	else:
		Input.action_release(neg)
		Input.action_release(pos)

func _reset_joystick() -> void:
	touch_id = -1
	is_touched = false
	knob_offset = Vector2.ZERO
	base_pos = _get_default_pos()
	for action in [&"ui_left", &"ui_right", &"ui_up", &"ui_down", jump_action]:
		Input.action_release(action)
	queue_redraw()

@export var action_name: StringName = &"ui_attack"

var touch_id: int = -1

	action_mode = BaseButton.ACTION_MODE_BUTTON_PRESS


func _gui_input(event: InputEvent) -> void:
	if event is InputEventScreenTouch:
		if event.pressed and touch_id == -1:
			touch_id = event.index
			_send_action(true)
		elif not event.pressed and event.index == touch_id:
			touch_id = -1
			_send_action(false)


func _send_action(is_pressed: bool) -> void:
	var ev = InputEventAction.new()
	ev.action = action_name
	ev.pressed = is_pressed
	Input.parse_input_event(ev)

enum State { PATROL, CHASE }

@export_category("Enemy")
@export var enemy_id: StringName

@export var enemy_sprite: Texture2D

@export_category("Stats")

@export var chase_stop_distance: float = 8.0

@export_category("Loot")
@export var loot: Array[PackedScene] = []

@export_category("Knockback")
@export var knockback_force: float = 90.0
@export var knockback_upward_force: float = 40.0
@export var knockback_time: float = 0.15

@onready var sprite: Sprite2D = $Sprite2D
@onready var floor_check: ShapeCast2D = $FloorCheck
@onready var detection_area: Area2D = $DetectionArea
@onready var damage_area: Area2D = $DamageArea

var player_in_damage_area: Node2D = null
var current_health: int
var state: State = State.PATROL
var move_direction: float = 1.0
var target: Node2D = null
var is_dead: bool = false
var is_knocked_back: bool = false

func _ready() -> void:
	current_health = max_health
	sprite.texture = enemy_sprite
	detection_area.body_entered.connect(func(b): if b.is_in_group("player"): target = b; state = State.CHASE)
	detection_area.body_exited.connect(func(b): if b == target: target = null; state = State.PATROL)
	damage_area.body_entered.connect(_on_damage_body_entered)
	damage_area.body_exited.connect(func(b): if b == player_in_damage_area: player_in_damage_area = null)
	Game.invulnerability_ended.connect(_try_damage_player)

func _physics_process(delta: float) -> void:
	if GameState.current_mode != &"platformer":
		velocity = Vector2.ZERO
		return

	if not is_on_floor():
		velocity.y += gravity * delta

	if is_knocked_back:
		move_and_slide()
		return

	match state:
		State.PATROL:
			if not _has_floor_ahead(move_direction) or is_on_wall():
				_set_direction(-move_direction)
			velocity.x = move_direction * speed

		State.CHASE:
			if not is_instance_valid(target):
				state = State.PATROL
			else:
				var dx := target.global_position.x - global_position.x
				if absf(dx) <= chase_stop_distance:
					velocity.x = 0.0
				else:
					var dir := signf(dx)
					_set_direction(dir)
					velocity.x = (dir * speed) if _has_floor_ahead(dir) else 0.0

	move_and_slide()

func _has_floor_ahead(dir: float) -> bool:
	floor_check.position.x = absf(floor_check.position.x) * dir
	floor_check.force_shapecast_update()
	return floor_check.is_colliding()

func _set_direction(dir: float) -> void:
	if is_zero_approx(dir): return
	move_direction = dir
	sprite.flip_h = dir < 0.0

func take_damage(amount: int) -> void:
	if is_dead or amount <= 0: return
	current_health = maxi(current_health - amount, 0)
	if current_health <= 0:
		die()

func die() -> void:
	if is_dead: return
	is_dead = true
	for loot_scene in loot:
		if loot_scene:
			var item = loot_scene.instantiate()
			get_parent().add_child(item)
			if item is Node2D:
				item.global_position = global_position
	queue_free()

func _on_damage_body_entered(body: Node2D) -> void:
	if body.is_in_group("player"):
		player_in_damage_area = body
		_try_damage_player()

func _try_damage_player() -> void:
	if not is_instance_valid(player_in_damage_area): return
	if Game.take_damage(damage):
		var dir := signf(player_in_damage_area.global_position.x - global_position.x)
		player_in_damage_area.knockback(Vector2(dir, -0.25).normalized())

func knockback(direction: Vector2) -> void:
	if is_dead: return
	is_knocked_back = true
	velocity = Vector2(direction.x * knockback_force, -knockback_upward_force)
	get_tree().create_timer(knockback_time).timeout.connect(func(): is_knocked_back = false)

func get_currency(currency_id: String) -> int:
	return currencies.get(currency_id, 0)

func spend_currency(currency_id: String, amount: int) -> bool:
	if amount <= 0 or get_currency(currency_id) < amount:
		return false
	currencies[currency_id] -= amount
	changed.emit()
	return true

func add_weapon(weapon: WeaponData) -> bool:
	if not weapon or weapon.weapon_id.is_empty() or weapons.has(weapon.weapon_id):
		return false
	weapons[weapon.weapon_id] = weapon
	if equipped_weapon_id.is_empty():
		equipped_weapon_id = weapon.weapon_id
	changed.emit()
	return true

func equip_weapon(weapon_id: StringName) -> bool:
	if not weapons.has(weapon_id):
		return false
	equipped_weapon_id = weapon_id
	changed.emit()
	return true

func get_equipped_weapon() -> WeaponData:
	return weapons.get(equipped_weapon_id)

func has_weapon(weapon_id: StringName) -> bool:
	return weapons.has(weapon_id)

func get_weapon(weapon_id: StringName) -> WeaponData:
	return weapons.get(weapon_id)

func clear() -> void:
	currencies.clear()
	weapons.clear()
	equipped_weapon_id = &""
	changed.emit()

func print_inventory() -> void:
	print("=== INVENTORY ===", "\nCurrencies: ", currencies, "\nWeapons: ", weapons.keys())

var time_begin: float = 0
var beat_duration: float = 0
var last_beat: int = 0

func _ready():
	beat_duration = 60.0 / bpm
	if music_track:
		audio_player.stream = music_track
	
	# Сами подписываемся на свой же сигнал для визуала
	beat.connect(_on_own_beat)

func play_music():
	time_begin = Time.get_ticks_usec()
	audio_player.play()

func _process(_delta):
	if audio_player.playing:
		var time = (Time.get_ticks_usec() - time_begin) / 1000000.0
		var current_beat = int(floor(time / beat_duration))
		
		if current_beat > last_beat:
			last_beat = current_beat
			beat.emit(last_beat)

func _on_own_beat(_index):
	flash_rect.color = Color.WHITE
	var tween = create_tween()
	tween.tween_property(flash_rect, "color", flash_color, 0.2)
